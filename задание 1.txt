#include <iostream> 
#include <cmath>    

/**
 * @brief Représente un point dans un plan 2D.
 *
 * Un point a une position définie par ses coordonnées 'x' et 'y'.
 */
struct Point {
    double x; 
    double y; 

    /**
     * @brief Crée un nouveau point.
     * @param x_val Valeur pour la coordonnée x (par défaut 0.0).
     * @param y_val Valeur pour la coordonnée y (par défaut 0.0).
     */
    Point(double x_val = 0.0, double y_val = 0.0) : x(x_val), y(y_val) {}

    /**
     * @brief Calcule la distance entre ce point et un autre point.
     * @param other L'autre point.
     * @return La distance entre les deux points.
     */
    double distanceTo(const Point& other) const {
        return std::sqrt(std::pow(x - other.x, 2) + std::pow(y - other.y, 2));
    }
};

/**
 * @brief Représente un triangle défini par trois points.
 *
 * Cette classe permet de créer un triangle et de calculer son périmètre et sa surface.
 */
class Triangle {
private:
    
    Point p1;
    Point p2;
    Point p3;

public:
    /**
     * @brief Crée un nouveau triangle avec trois points donnés.
     * @param point1 Le premier sommet.
     * @param point2 Le deuxième sommet.
     * @param point3 Le troisième sommet.
     */
    Triangle(const Point& point1, const Point& point2, const Point& point3)
        : p1(point1), p2(point2), p3(point3) {
    }

    /**
     * @brief Donne le premier point du triangle.
     * @return Le premier point.
     */
    Point getP1() const { return p1; }

    /**
     * @brief Donne le deuxième point du triangle.
     * @return Le deuxième point.
     */
    Point getP2() const { return p2; }

    /**
     * @brief Donne le troisième point du triangle.
     * @return Le troisième point.
     */
    Point getP3() const { return p3; }

    /**
     * @brief Calcule le périmètre du triangle.
     *
     * Le périmètre est la somme des longueurs des trois côtés du triangle.
     */
    double calculatePerimeter() const {
        double cote1 = p1.distanceTo(p2); 
        double cote2 = p2.distanceTo(p3); 
        double cote3 = p3.distanceTo(p1); 
        return cote1 + cote2 + cote3;
    }

    /**
     * @brief Calcule la surface (l'aire) du triangle.
     *
     * Utilise la formule des coordonnées (formule de laçage) qui est très pratique
     * quand on connaît les coordonnées des sommets.
     */
    double calculateArea() const {
        double aire = 0.5 * std::fabs(
            p1.x * (p2.y - p3.y) +
            p2.x * (p3.y - p1.y) +
            p3.x * (p1.y - p2.y)
        );
        return aire;
    }
};

/**
 * @brief Fonction principale du programme.
 *
 * C'est ici que le programme commence. On va créer des triangles et tester nos fonctions.
 */
int main() {
    Point ptA(0, 0);
    Point ptB(3, 0);
    Point ptC(0, 4);

    Triangle monTriangle1(ptA, ptB, ptC);

    std::cout << "--- Triangle 1 (Points: (0,0), (3,0), (0,4)) ---\n";
    std::cout << "  Perimetre : " << monTriangle1.calculatePerimeter() << std::endl; 
    std::cout << "  Surface   : " << monTriangle1.calculateArea() << std::endl;     
    std::cout << std::endl;

    Point ptD(1, 1);
    Point ptE(4, 5);
    Point ptF(7, 1);
    Triangle monTriangle2(ptD, ptE, ptF);

    std::cout << "--- Triangle 2 (Points: (1,1), (4,5), (7,1)) ---\n";
    std::cout << "  Perimetre : " << monTriangle2.calculatePerimeter() << std::endl; // Environ 16
    std::cout << "  Surface   : " << monTriangle2.calculateArea() << std::endl;     // Doit être 12
    std::cout << std::endl;
    Point ptG(0, 0);
    Point ptH(1, 1);
    Point ptI(2, 2);
    Triangle monTriangle3(ptG, ptH, ptI);

    std::cout << "--- Triangle 3 (Points alignes: (0,0), (1,1), (2,2)) ---\n";
    std::cout << "  Perimetre : " << monTriangle3.calculatePerimeter() << std::endl; // Somme des distances, comme s'ils formaient une ligne
    std::cout << "  Surface   : " << monTriangle3.calculateArea() << std::endl;     // Doit être 0 ou très proche de 0
    std::cout << "  (Si la surface est 0, ca veut dire que les points sont alignes!)\n";
    std::cout << std::endl;

    return 0; 
}
